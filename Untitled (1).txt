local Compat = {}

function Compat.getExecutor()
    if rawget(getfenv(), "Delta") or rawget(getfenv(), "delta") then return "Delta" end
    if identifyexecutor then
        local ok, name = pcall(identifyexecutor)
        if ok then return name end
    end
    return "Unknown"
end

function Compat.getHui()
    local ok, gui = pcall(function()
        if gethui then return gethui() end
        if get_hidden_gui then return get_hidden_gui() end
        return Player:WaitForChild("PlayerGui", 5)
    end)
    if ok and gui then return gui end
    return Player:WaitForChild("PlayerGui")
end

function Compat.request(options)
    if request then return request(options) end
    if http_request then return http_request(options) end
    if syn and syn.request then return syn.request(options) end
    if http and http.request then return http.request(options) end
    return nil
end

function Compat.Protect(fn, ...)
    local args = {...}
    return pcall(function() return fn(table.unpack(args)) end)
end

function Compat.isMobile()
    return UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
end

function Compat.isExecutorReady()
    return Player ~= nil and Workspace ~= nil and Camera ~= nil
end

-- Шлях збереження конфігу
local CONFIG_FOLDER = "CatnapHub_Delta_v2"
local CONFIG_FILE = CONFIG_FOLDER .. "/config.json"

local function SaveConfig()
    pcall(function()
        if writefile and isfolder and not isfolder(CONFIG_FOLDER) then
            makefolder(CONFIG_FOLDER)
        end
        if writefile then
            local saveData = {}
            for k, v in pairs(Config) do
                if type(v) == "Color3" then
                    saveData[k] = { __type = "Color3", r = v.R, g = v.G, b = v.B }
                elseif type(v) == "Vector3" then
                    saveData[k] = { __type = "Vector3", x = v.X, y = v.Y, z = v.Z }
                elseif type(v) ~= "table" and type(v) ~= "function" and type(v) ~= "userdata" then
                    saveData[k] = v
                elseif type(v) == "table" and k == "Waypoints" then
                    saveData[k] = v
                end
            end
            writefile(CONFIG_FILE, HttpService:JSONEncode(saveData))
        end
    end)
end

local function LoadConfig()
    pcall(function()
        if isfile and isfile(CONFIG_FILE) and readfile then
            local data = HttpService:JSONDecode(readfile(CONFIG_FILE))
            for k, v in pairs(data) do
                if type(v) == "table" and v.__type == "Color3" then
                    Config[k] = Color3.new(v.r, v.g, v.b)
                elseif type(v) == "table" and v.__type == "Vector3" then
                    Config[k] = Vector3.new(v.x, v.y, v.z)
                else
                    Config[k] = v
                end
            end
        end
    end)
end

LoadConfig()

local Utils = {}

function Utils.getCharacter()
    return Player.Character
end

function Utils.getHumanoid()
    local char = Utils.getCharacter()
    if char then return char:FindFirstChildOfClass("Humanoid") end
    return nil
end

function Utils.getRoot()
    local char = Utils.getCharacter()
    if char then return char:FindFirstChild("HumanoidRootPart") end
    return nil
end

function Utils.getPing()
    local ok, ping = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    if ok then return math.floor(ping) end
    return 0
end

function Utils.getFPS()
    local ok, fps = pcall(function()
        return math.floor(1 / RunService.RenderStepped:Wait())
    end)
    return fps or 60
end

function Utils.distance(a, b)
    return (a - b).Magnitude
end

function Utils.formatTime(t)
    local hours = math.floor(t / 3600)
    local mins = math.floor((t % 3600) / 60)
    local secs = math.floor(t % 60)
    return string.format("%02d:%02d:%02d", hours, mins, secs)
end

function Utils.randomString(len)
    local chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
    local s = ""
    for i = 1, len do
        local idx = math.random(1, #chars)
        s = s .. chars:sub(idx, idx)
    end
    return s
end

function Utils.isAlive(plr)
    if not plr or not plr.Character then return false end
    local hum = plr.Character:FindFirstChildOfClass("Humanoid")
    return hum and hum.Health > 0
end

function Utils.getAlivePlayers()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player and Utils.isAlive(p) then
            table.insert(list, p)
        end
    end
    return list
end

function Utils.getAlivePlayersCount()
    return #Utils.getAlivePlayers()
end

function Utils.notify(title, text, variant)
    if not Config.NotificationsEnabled then return end
    pcall(function()
        if _G.CATNAP_UI and _G.CATNAP_UI.Notify then
            _G.CATNAP_UI.Notify:Push({
                Title = tostring(title),
                Content = tostring(text),
                Variant = variant or "Info",
                Duration = 3,
            })
        end
    end)
end

function Utils.playSound(id)
    pcall(function()
        local sound = Instance.new("Sound")
        sound.SoundId = "rbxassetid://" .. tostring(id)
        sound.Volume = 0.5
        sound.Parent = SoundService
        sound:Play()
        task.delay(3, function() sound:Destroy() end)
    end)
end

ESP = { Objects = {} }

function ESP:Create(player)
    if player == Player or ESP.Objects[player] then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    -- Highlight
    local highlight = Instance.new("Highlight")
    highlight.Name = "CATNAP_ESP"
    highlight.FillColor = Config.ESP_Color
    highlight.OutlineColor = Config.ESP_Color
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0.1
    highlight.Parent = char

    -- Billboard
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "CATNAP_ESP_BB"
    billboard.Size = UDim2.new(0, 220, 0, 60)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Parent = root

    local nameLabel = Instance.new("TextLabel")
    nameLabel.Name = "NameLabel"
    nameLabel.Size = UDim2.new(1, 0, 0.4, 0)
    nameLabel.BackgroundTransparency = 1
    nameLabel.Text = player.Name
    nameLabel.TextColor3 = Color3.new(1, 1, 1)
    nameLabel.TextStrokeTransparency = 0
    nameLabel.TextScaled = true
    nameLabel.Font = Enum.Font.GothamBold
    nameLabel.Parent = billboard

    local distLabel = Instance.new("TextLabel")
    distLabel.Name = "DistLabel"
    distLabel.Size = UDim2.new(1, 0, 0.3, 0)
    distLabel.Position = UDim2.new(0, 0, 0.4, 0)
    distLabel.BackgroundTransparency = 1
    distLabel.Text = "0"
    distLabel.TextColor3 = Color3.fromRGB(255, 255, 0)
    distLabel.TextStrokeTransparency = 0
    distLabel.TextScaled = true
    distLabel.Font = Enum.Font.Gotham
    distLabel.Parent = billboard

    local hpLabel = Instance.new("TextLabel")
    hpLabel.Name = "HpLabel"
    hpLabel.Size = UDim2.new(1, 0, 0.3, 0)
    hpLabel.Position = UDim2.new(0, 0, 0.7, 0)
    hpLabel.BackgroundTransparency = 1
    hpLabel.Text = "100"
    hpLabel.TextColor3 = Color3.fromRGB(0, 255, 0)
    hpLabel.TextStrokeTransparency = 0
    hpLabel.TextScaled = true
    hpLabel.Font = Enum.Font.Gotham
    hpLabel.Parent = billboard

    ESP.Objects[player] = {
        highlight = highlight,
        billboard = billboard,
        nameLabel = nameLabel,
        distLabel = distLabel,
        hpLabel = hpLabel,
    }
end

function ESP:Remove(player)
    if ESP.Objects[player] then
        for _, obj in pairs(ESP.Objects[player]) do
            if obj and obj.Parent then obj:Destroy() end
        end
        ESP.Objects[player] = nil
    end
end

function ESP:ClearAll()
    for p, _ in pairs(ESP.Objects) do ESP:Remove(p) end
end

function ESP:UpdateColor(color)
    for _, data in pairs(ESP.Objects) do
        if data.highlight then
            data.highlight.FillColor = color
            data.highlight.OutlineColor = color
        end
    end
end

function ESP:Update()
    if not Config.ESP_Enabled then return end
    local myRoot = Utils.getRoot()
    for player, data in pairs(ESP.Objects) do
        if player.Character then
            local hum = player.Character:FindFirstChildOfClass("Humanoid")
            local root = player.Character:FindFirstChild("HumanoidRootPart")
            if hum and root then
                if Config.ESP_ShowDistance and myRoot then
                    data.distLabel.Text = math.floor(Utils.distance(myRoot.Position, root.Position)) .. "m"
                else
                    data.distLabel.Text = ""
                end
                if Config.ESP_ShowHealth then
                    data.hpLabel.Text = math.floor(hum.Health) .. " HP"
                    local ratio = hum.Health / hum.MaxHealth
                    data.hpLabel.TextColor3 = Color3.fromRGB(255 * (1 - ratio), 255 * ratio, 0)
                else
                    data.hpLabel.Text = ""
                end
                data.nameLabel.Text = Config.ESP_ShowName and player.Name or ""
            end
        end
    end
end

-- БЛОК 19: СИСТЕМА AIMBOT
--=====================================================================
local Aimbot = { Target = nil, LastUpdate = 0 }

function Aimbot:GetClosest()
    local closest, shortest = nil, math.huge
    if not Camera then return nil end
    local vp = Camera.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local part = p.Character:FindFirstChild(Config.Aimbot_Part)
            if hum and hum.Health > 0 and part then
                local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local dist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if dist < shortest and dist <= Config.Aimbot_FOV then
                        shortest = dist
                        closest = p
                    end
                end
            end
        end
    end
    return closest
end

function Aimbot:Update()
    if not Config.Aimbot_Enabled then return end
    if not Camera then return end
    local target = Aimbot:GetClosest()
    if target and target.Character then
        local part = target.Character:FindFirstChild(Config.Aimbot_Part)
        if part then
            local cur = Camera.CFrame
            local new = CFrame.new(cur.Position, part.Position)
            Camera.CFrame = cur:Lerp(new, Config.Aimbot_Smooth)
        end
    end
end

local TriggerBot = { LastShot = 0 }

function TriggerBot:Update()
    if not Config.TriggerBot_Enabled then return end
    local now = tick()
    if now - TriggerBot.LastShot < Config.TriggerBot_Delay then return end
    if not Camera then return end
    local vp = Camera.ViewportSize
    local center = Vector2.new(vp.X / 2, vp.Y / 2)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local part = p.Character:FindFirstChild("Head")
            if hum and hum.Health > 0 and part then
                local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
                if onScreen and (Vector2.new(sp.X, sp.Y) - center).Magnitude < 30 then
                    TriggerBot.LastShot = now
                    pcall(function()
                        VirtualUser:CaptureController()
                        VirtualUser:ClickButton1(Vector2.new(center.X, center.Y))
                    end)
                    break
                end
            end
        end
    end
end

local KillAura = { LastHit = 0 }

function KillAura:Update()
    if not Config.KillAura_Enabled then return end
    local now = tick()
    if now - KillAura.LastHit < 0.2 then return end
    local myRoot = Utils.getRoot()
    if not myRoot then return end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and root then
                if Utils.distance(myRoot.Position, root.Position) < Config.KillAura_Range then
                    KillAura.LastHit = now
                    pcall(function() hum:TakeDamage(5) end)
                end
            end
        end
    end
end

-- БЛОК 22: СИСТЕМА FLY
--=====================================================================
local Fly = { BV = nil, BG = nil }

function Fly:Stop()
    local char = Utils.getCharacter()
    if not char then return end
    if Fly.BV then Fly.BV:Destroy() Fly.BV = nil end
    if Fly.BG then Fly.BG:Destroy() Fly.BG = nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root then root.Velocity = Vector3.new(0, 0, 0) end
end

function Fly:Update()
    if not Config.Fly_Enabled then
        if Fly.BV then Fly:Stop() end
        return
    end
    local char = Utils.getCharacter()
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    if Config.Fly_Mode == "Velocity" then
        if not Fly.BV then
            Fly.BV = Instance.new("BodyVelocity")
            Fly.BV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
            Fly.BV.Velocity = Vector3.new(0, 0, 0)
            Fly.BV.Parent = root
        end

        local dir = Vector3.new(0, 0, 0)
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0, 1, 0) end

        if dir.Magnitude > 0 then
            Fly.BV.Velocity = dir.Unit * Config.Fly_Speed
        else
            Fly.BV.Velocity = Vector3.new(0, 0, 0)
        end
    elseif Config.Fly_Mode == "CFrame" then
        if Fly.BV then Fly.BV:Destroy() Fly.BV = nil end
        local dir = Vector3.new(0, 0, 0)
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0, 1, 0) end
        if dir.Magnitude > 0 then
            root.CFrame = root.CFrame + dir.Unit * (Config.Fly_Speed / 60)
        end
    end
end
