--============================================================

--============================================================

local function notify(title, text, variant)
    if not Config.Notifications then
        return
    end

    pcall(function()
        UI.Notify:Push({
            Title = title,
            Content = text,
            Variant = variant or "Info",
            Duration = 3,
        })
    end)
end

local function getPing()
    local success, result = pcall(function()
        return Stats.Network.ServerStatsItem
            ["Data Ping"]
            :GetValue()
    end)

    if success then
        return math.floor(result)
    end

    return 0
end

--============================================================
-- LIVE DEVICE UPDATE
--============================================================

local lastViewport = ""

RunService.RenderStepped:Connect(function()

    local camera =
        workspace.CurrentCamera

    if not camera then
        return
    end

    local viewport =
        camera.ViewportSize

    local viewportKey =
        tostring(
            math.floor(viewport.X)
        )
        .. "x"
        .. tostring(
            math.floor(viewport.Y)
        )

    if viewportKey ~= lastViewport then

        lastViewport = viewportKey

        pcall(function()

            if UI.Device then

                if UI.Device.Class then
                    UI.State:Set(
                        "DeviceClass",
                        tostring(
                            UI.Device.Class
                        )
                    )
                end

                if UI.Device.Layout then
                    UI.State:Set(
                        "DeviceLayout",
                        tostring(
                            UI.Device.Layout
                        )
                    )
                end

            end

        end)

    end

end)

--============================================================
-- LIVE PING
--============================================================

task.spawn(function()

        task.wait(1)

        local ping =
            getPing()

        pcall(function()

            UI.State:Set(
                "Ping",
                tostring(ping) .. " ms"
            )

        end)

    end

end)
